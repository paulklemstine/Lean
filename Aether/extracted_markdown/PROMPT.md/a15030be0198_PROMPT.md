/-!
# Fibonacci as a Strong Divisibility Sequence

This file develops, from first principles, the divisibility calculus of the Fibonacci
sequence built on the single structural identity

  `Nat.fib_gcd : Nat.fib (Nat.gcd m n) = Nat.gcd (Nat.fib m) (Nat.fib n)`

(the "Fib_gcd_identity" of the catalog, cf.
`Catalog/Algebra/Tropical_..._Fibonacci_Primitive_Divisors.lean`).  From it we obtain:

* `fib_dvd_gcd_indices` — the index set `{k | d ∣ fib k}` is closed under `gcd`;
* `fib_dvd_fib_iff` — for `3 ≤ m`, `fib m ∣ fib n ↔ m ∣ n` (full divisibility characterization);
* `fib_coprime_iff` — `Coprime (fib m) (fib n) ↔ gcd m n = 1 ∨ gcd m n = 2`;
* `fib_dvd_iff_fibRank_dvd` — the **law of apparition** for an *arbitrary* modulus `d`
  (no primality hypothesis), generalizing the catalog's prime-only
  `fib_dvd_iff_entry_dvd`.

These are the elementary building blocks of the theory of Lucas/Fibonacci primitive
prime divisors used in the catalog's Carmichael development
(`Catalog/Shared/CarmichaelProof.lean`, `fib_carmichael_composite`).

-- !-- Lab Notebook -- !--
Hypothesis: Every divisibility fact about `Nat.fib` should reduce, with no case analysis,
  to the gcd identity `fib (gcd m n) = gcd (fib m) (fib n)` plus the fact that `fib` is
  strictly monotone on `Set.Ici 2` (hence injective there).
Result: All four theorems below follow from exactly these two inputs; primality is never
  needed for the law of apparition (it was an unnecessary hypothesis in the catalog version).
Insight: "Strong divisibility sequence" (`gcd a_m a_n = a_{gcd m n}`) is the right
  abstraction: divisibility of values mirrors divisibility of indices, modulo the two
  degenerate indices `1, 2` where `fib` fails to be injective (`fib 1 = fib 2 = 1`).
Failure analysis: The naive statement `fib m ∣ fib n ↔ m ∣ n` is FALSE at `m = 1, 2`
  (`fib 1 = fib 2 = 1` divides everything); the hypothesis `3 ≤ m` is exactly what
  restores injectivity of `fib` and makes the characterization sharp.
-/

open Nat

namespace FibStrongDivisibility

/-! ## 1. Closure of the index set under `gcd` -/

-- !-- `d ∣ fib m` and `d ∣ fib n` give `d ∣ gcd (fib m) (fib n) = fib (gcd m n)` by
--     the Fibonacci gcd identity; closure under `gcd` of indices is immediate. -- !--
theorem fib_dvd_gcd_indices (d m n : ℕ) (hm : d ∣ Nat.fib m) (hn : d ∣ Nat.fib n) :
    d ∣ Nat.fib (Nat.gcd m n) := by
  rw [ Nat.fib_gcd ];
  exact Nat.dvd_gcd hm hn

/-! ## 2. Full divisibility characterization -/

-- !-- (←) is `Nat.fib_dvd`. (→): `fib m ∣ fib n` gives `gcd (fib m) (fib n) = fib m`,
--     i.e. `fib (gcd m n) = fib m`; since `m ≥ 3` forces `gcd m n ≥ 2`, injectivity of
--     `fib` on `Set.Ici 2` yields `gcd m n = m`, i.e. `m ∣ n`. -- !--
theorem fib_dvd_fib_iff (m n : ℕ) (hm : 3 ≤ m) :
    Nat.fib m ∣ Nat.fib n ↔ m ∣ n := by
  refine' ⟨ fun h => _, fun h => _ ⟩;
  · -- Since `fib m ∣ fib n`, we have `gcd (fib m) (fib n) = fib m`.
    have h_gcd : Nat.gcd (Nat.fib m) (Nat.fib n) = Nat.fib m := by
      exact Nat.gcd_eq_left h;
    -- By `Nat.fib_gcd`, `Nat.fib (Nat.gcd m n) = Nat.fib m`.
    have h_fib_gcd : Nat.fib (Nat.gcd m n) = Nat.fib m := by
      rw [ ← h_gcd, Nat.fib_gcd ];
    have h_gcd_ge_two : 2 ≤ Nat.gcd m n := by
      contrapose! h_fib_gcd; interval_cases _ : Nat.gcd m n <;> simp_all +decide ;
      linarith [ Nat.le_fib_add_one m ];
    exact Nat.gcd_eq_left_iff_dvd.mp ( StrictMonoOn.injOn ( show StrictMonoOn Nat.fib ( Set.Ici 2 ) from Nat.fib_strictMonoOn ) ( show 2 ≤ Nat.gcd m n from h_gcd_ge_two ) ( show 2 ≤ m from by linarith ) h_fib_gcd );
  · obtain ⟨ k, rfl ⟩ := h; simp +decide [ Nat.fib_dvd ] ;

/-! ## 3. Coprimality characterization -/

-- !-- `Coprime (fib m) (fib n) ↔ gcd (fib m) (fib n) = 1 ↔ fib (gcd m n) = 1`, and
--     `fib k = 1 ↔ k = 1 ∨ k = 2`. -- !--
theorem fib_coprime_iff (m n : ℕ) :
    Nat.Coprime (Nat.fib m) (Nat.fib n) ↔ Nat.gcd m n = 1 ∨ Nat.gcd m n = 2 := by
  rw [ Nat.coprime_iff_gcd_eq_one, ← Nat.fib_gcd ];
  rcases k : Nat.gcd m n with ( _ | _ | _ | k ) <;> simp_all +arith +decide [ Nat.fib_add_two ];
  linarith [ Nat.fib_pos.2 ( Nat.succ_pos ‹_› ) ]

/-! ## 4. The Fibonacci rank of apparition and the law of apparition

For a modulus `d`, the *rank of apparition* `fibRank d` is the least positive `k`
with `d ∣ fib k` (and `0` if none exists).  The law of apparition states that the
indices `n` with `d ∣ fib n` are *exactly* the multiples of `fibRank d`.  Unlike the
catalog's `fib_dvd_iff_entry_dvd`, no primality of `d` is required. -/

open Classical in
/-- The rank of apparition of `d` in the Fibonacci sequence: least `k > 0` with `d ∣ fib k`. -/
noncomputable def fibRank (d : ℕ) : ℕ :=
  if h : ∃ k : ℕ, 0 < k ∧ d ∣ Nat.fib k then Nat.find h else 0

theorem fibRank_pos (d : ℕ) (hex : ∃ k : ℕ, 0 < k ∧ d ∣ Nat.fib k) :
    0 < fibRank d := by
  unfold fibRank; aesop;

theorem fibRank_dvd (d : ℕ) (hex : ∃ k : ℕ, 0 < k ∧ d ∣ Nat.fib k) :
    d ∣ Nat.fib (fibRank d) := by
  convert Nat.find_spec hex |>.2 using 1;
  unfold fibRank; aesop;

-- !-- (←) `fibRank d ∣ n` gives `fib (fibRank d) ∣ fib n` (Nat.fib_dvd), and `d ∣
--     fib (fibRank d)`, so `d ∣ fib n`. (→): if `d ∣ fib n` then by gcd-closure
--     `d ∣ fib (gcd (fibRank d) n)`; minimality of `fibRank d` forces
--     `gcd (fibRank d) n = fibRank d`, i.e. `fibRank d ∣ n`. -- !--
theorem fib_dvd_iff_fibRank_dvd (d n : ℕ) (hn : 0 < n)
    (hex : ∃ k : ℕ, 0 < k ∧ d ∣ Nat.fib k) :
    d ∣ Nat.fib n ↔ fibRank d ∣ n := by
  constructor <;> intro h;
  · have h_gcd : Nat.gcd (fibRank d) n = fibRank d := by
      refine' Nat.le_antisymm _ _;
      · exact Nat.le_of_dvd ( Nat.pos_of_ne_zero ( by unfold fibRank; aesop ) ) ( Nat.gcd_dvd_left _ _ );
      · have h_gcd : d ∣ Nat.fib (Nat.gcd (fibRank d) n) := by
          exact fib_dvd_gcd_indices d _ _ ( fibRank_dvd d hex ) h;
        unfold fibRank at *;
        split_ifs at * ; simp_all +decide;
        exact ⟨ _, le_rfl, Nat.gcd_pos_of_pos_left _ ( Nat.find_spec hex |>.1 ), h_gcd ⟩;
    exact h_gcd ▸ Nat.gcd_dvd_right _ _;
  · exact dvd_trans ( fibRank_dvd d hex ) ( by obtain ⟨ k, rfl ⟩ := h; simp [ Nat.fib_dvd ] )

end FibStrongDivisibility

### CORE DEFINITIONS (5+ required)

```lean
/-- The closure fiber operator: given a classifier f on X,
    cl_f(A) = f⁻¹(f(A)) = {x : ∃ y ∈ A, f x = f y}.
    This is the union of all label-fibers intersecting A.
    Bridge: connects set-theoretic closure to classification boundaries. -/
def closureFiber {X C : Type*} [DecidableEq C] (f : X → C) (A : Set X) : Set X :=
  {x : X | ∃ y ∈ A, f x = f y}

/-- A closure classifier bundles a classifier with its Lipschitz constant
    and the induced EML closure operator. The Lipschitz constant L certifies
    that the decision boundary lies at distance ≥ 1/L from any interior point
    of a fiber (using the discrete metric on labels). -/
structure ClosureClassifier (X C : Type*) [PseudoMetricSpace X] [DecidableEq C] where
  classify : X → C
  lipschitzConstant : ℝ
  lipschitz_certified : LipschitzWith lipschitzConstant classify

/-- The certified robustness radius at a point x under classifier f:
    r_f(x) = inf{d(x,y) : f y ≠ f x} = d(x, ∂(closureFiber f {x})).
    This is the distance to the nearest decision boundary. -/
def certifiedRadius {X C : Type*} [PseudoMetricSpace X] [DecidableEq C]
    (f : X → C) (x : X) : ℝ :=
  sInf {r : ℝ | ∃ y : X, dist x y ≤ r ∧ f y ≠ f x}

/-- The lattice height of a finite lattice: the length of the longest
    strictly increasing chain. This bounds the VC-dimension of the
    closure classifier family. -/
def latticeHeight {α : Type*} [PartialOrder α] [Finite α] : ℕ :=
  (Finset.univ : Finset α).chains.chainLength  -- maximum chain length

/-- Iteration of a monotone extensive operator toward its fixed point.
    For an EML closure operator, idempotence guarantees convergence in
    exactly 1 step; for a merely monotone extensive operator on a lattice
    of height h, convergence in at most h steps. -/
def iterateTowardFixedPoint {α : Type*} [CompleteLattice α]
    (φ : α → α) (h_mono : Monotone φ) (n : ℕ) : α :=
  (Finset.range n).foldr (fun _ => φ) ⊥
```

### THEOREM 1: Closure Fiber is an EML Operator (The Algebraic Foundation)

```lean
/-- Extensivity: every point in A belongs to its own fiber closure.
    Proof: for x ∈ A, take y = x, then f x = f x. -/
theorem closureFiber_extensive {X C : Type*} [DecidableEq C]
    (f : X → C) (A : Set X) :
    A ⊆ closureFiber f A := by
  intro x hx; exact ⟨x, hx, rfl⟩

/-- Monotonicity: enlarging the set can only enlarge the fiber closure.
    Proof: if x ∈ cl_f(A) via y ∈ A, and A ⊆ B, then y ∈ B. -/
theorem closureFiber_monotone {X C : Type*} [DecidableEq C]
    (f : X → C) {A B : Set X} (hAB : A ⊆ B) :
    closureFiber f A ⊆ closureFiber f B := by
  intro x ⟨y, hyA, hyx⟩; exact ⟨y, hAB hyA, hyx⟩

/-- Idempotence: applying the fiber closure twice is the same as once.
    Proof: cl_f(cl_f(A)) = {x : ∃z ∈ cl_f(A), f x = f z}
         = {x : ∃z, ∃y ∈ A, f z = f y ∧ f x = f z}
         = {x : ∃y ∈ A, f x = f y} = cl_f(A). -/
theorem closureFiber_idempotent {X C : Type*} [DecidableEq C]
    (f : X → C) (A : Set X) :
    closureFiber f (closureFiber f A) = closureFiber f A := by
  ext x; constructor
  · rintro ⟨z, ⟨y, hyA, hyz⟩, hxz⟩
    exact ⟨y, hyA, hxz.trans hyz⟩
  · intro hx; exact ⟨x, hx, rfl⟩

/-- Master theorem: closureFiber is an EML closure operator.
    This is the algebraic foundation of closure-theoretic ML. -/
theorem closureFiber_eml_operator {X C : Type*} [DecidableEq C]
    (f : X → C) :
    IsClosureOperator (closureFiber f) := by
  -- Construct from the three properties above
  exact { extensive := closureFiber_extensive f
        , monotone := fun _ _ => closureFiber_monotone f
        , idempotent := fun _ => closureFiber_idempotent f _ }

/-- The fiber of a single point is its entire label-class.
    This connects the closure operator to the decision boundary. -/
theorem closureFiber_singleton_eq_fiber {X C : Type*} [DecidableEq C]
    (f : X → C) (x : X) :
    closureFiber f {x} = {y : X | f y = f x} := by
  ext y; constructor
  · rintro ⟨z, hz, hyz⟩; rw [Set.mem_singleton_iff] at hz; subst hz; exact hyz
  · intro hy; exact ⟨x, Set.mem_singleton x, hy⟩
```

```lean
    certifiedRadius f x = sInf {d : ℝ | ∃ y ∉ closureFiber f {x}, d = dist x y} := by
  -- Key step: f y ≠ f x ↔ y ∉ cl_f({x}) by closureFiber_singleton_eq_fiber
  -- Then the two infima range over the same set

    have : y ∉ closureFiber f {x} := by
      intro h; rw [closureFiber_singleton_eq_fiber] at h
      exact hy_ne h
    exact ⟨y, this, hy_bound⟩
  · rintro ⟨y, hy_nmem, hy_bound⟩
    have : f y ≠ f x := by
      intro h; apply hy_nmem
      rw [closureFiber_singleton_eq_fiber]; exact h
    exact ⟨y, hy_bound, this⟩

  -- EMetric.d (f x) (f y) ≤ L · d(x,y) < 1.
  -- But in the discrete metric, d(a,b) = 1 if a ≠ b, so f x = f y.
  -- Thus any y with f y ≠ f x must have d(x,y) ≥ 1/L.
  -- Hence certifiedRadius f x ≥ 1/L.
  rw [certifiedRadius]
  apply sInf_le_sInf
  intro r hr
  obtain ⟨y, hy_dist, hy_ne⟩ := hr

    simp [EMetric.d, Discrete.metric] at this
    -- In discrete metric: d < 1 means d = 0 means f x = f y
    exact absurd rfl hy_ne
  exact ⟨y, by linarith, hy_ne⟩

    decrease the certified radius. This follows from monotonicity of the
    closure fiber operator. -/

  -- ≥ certifiedRadius f x - d(x,y)

/-- The closure fiber of a Lipschitz classifier contains a metric ball of
    radius 1/L. This is the geometric content of certified robustness. -/
theorem closureFiber_contains_ball {X : Type*}
    [PseudoMetricSpace X] {C : Type*} [DecidableEq C] [Nonempty X]
    {f : X → C} {L : ℝ} (hL : LipschitzWith L f) (hL_pos : 0 < L) (x : X) :
    Metric.ball x (1/L) ⊆ closureFiber f {x} := by
```

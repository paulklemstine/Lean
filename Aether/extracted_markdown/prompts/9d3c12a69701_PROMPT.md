/-- A backpropagation derivation on a weight algebra: the map D(w) = -∂L/∂w

structure BackpropDerivation (R : Type*) [CommRing R] extends Derivation R R where
  loss_fn : R → R
  learning_rate : R
  deriv_eq_neg_grad : ∀ w, toFun w = -learning_rate * loss_fn w

/-- A training differential equation: the ODE dw/dt = D(w) governing
    weight evolution under gradient descent. -/
structure TrainingDiffEq (K : Type*) [Field K] (n : ℕ) where
  weight_algebra : WeightAlgebra K n
  derivation : BackpropDerivation (WeightAlgebra K n).carrier
  initial_weights : Fin n → K

/-- The differential Galois group of a training equation: automorphisms
    of the Picard-Vessiot extension commuting with the derivation.
    Classifies weight symmetries of the architecture. -/
structure DiffGaloisGroup (K : Type*) [Field K] (T : TrainingDiffEq K n) where
  carrier : Set (WeightAlgebra K n →ₐ[K] WeightAlgebra K n)
  commutes_with_deriv : ∀ σ ∈ carrier, ∀ w, σ (T.derivation.toFun w) = T.derivation.toFun (σ w)

/-- Ritt decomposition of a differential polynomial: factorization into
    irreducible components with bounded length. -/
structure RittDecomposition (R : Type*) [CommRing R] [Derivation R R] where
  components : List { p : DifferentialPolynomial R // Irreducible p }
  length_eq_components : components.length = rittLength p
  product_eq_original : components.map (·.val) |>.prod = p

/-!
# Perfect secrecy of the one-time pad over a finite group

This file formalizes the perfect secrecy of the one-time pad (OTP) when keys, messages
and ciphertexts live in an arbitrary finite group `G` and encryption is the group
multiplication `c = k * m`.

* `otp_unique_key`: for every message `m` and ciphertext `c` there is a *unique* key `k`
  with `k * m = c` (namely `k = c * m⁻¹`).
* `otp_key_cardinality`: exactly one key in `G` maps a given message to a given ciphertext.
* `otp_perfect_secrecy`: if the key is drawn uniformly at random from `G` (independently of
  the message), then observing the ciphertext gives no information about the message, i.e. the
  conditional probability `P(M = m | C = c)` equals the prior `P(M = m)`.

The uniform key forces the ciphertext to be uniform regardless of the message distribution,
which is the content of the Bayesian computation in `otp_perfect_secrecy`.

(Note: the requested import `Mathlib.Algebra.Group.Fintype` does not exist in this Mathlib
version; the finite-group instances used here are already available from the imports above,
and `Mathlib.Probability.Distributions.Uniform` is added to supply the uniform PMF.)
-/

open PMF
open scoped ENNReal

/-- For any message `m` and ciphertext `c` in a group, there is a unique key `k` with
`k * m = c`, namely `k = c * m⁻¹`. -/
theorem otp_unique_key (G : Type*) [Group G] [Fintype G] [DecidableEq G] (m c : G) :
    ∃! k : G, k * m = c :=
  ⟨c * m⁻¹, by show (c * m⁻¹) * m = c; rw [mul_assoc, inv_mul_cancel, mul_one],
    fun k hk => by rw [← hk, mul_assoc, mul_inv_cancel, mul_one]⟩

/-- Exactly one key `k` in `G` satisfies `k * m = c`. -/
theorem otp_key_cardinality (G : Type*) [Group G] [Fintype G] [DecidableEq G] (m c : G) :
    Finset.card (Finset.filter (fun k => k * m = c) Finset.univ) = 1 := by
  obtain ⟨k, hk, huniq⟩ := otp_unique_key G m c
  rw [Finset.card_eq_one]
  refine ⟨k, ?_⟩
  ext x
  simp only [Finset.mem_filter, Finset.mem_univ, true_and, Finset.mem_singleton]
  exact ⟨fun h => huniq x h, fun h => h ▸ hk⟩

/-- **Perfect secrecy of the one-time pad.**

Let the message be distributed according to an arbitrary PMF `messageDist` and let the key be
drawn uniformly and independently from the finite group `G`. The ciphertext is `C = K * M`,
so the joint distribution of message and ciphertext is

`joint = messageDist.bind (fun m' => (uniformOfFintype G).map (fun k => (m', k * m')))`.

Then the conditional probability of the message given the ciphertext,
`P(M = m | C = c) = joint (m, c) / P(C = c)`, equals the prior `P(M = m) = messageDist m`.
Here `P(C = c) = ∑' x, joint (x, c)` is the marginal probability of the ciphertext. -/
theorem otp_perfect_secrecy (G : Type*) [Group G] [Fintype G] [DecidableEq G]
    (messageDist : PMF G) (m c : G) :
    let joint : PMF (G × G) :=
      messageDist.bind (fun m' => (PMF.uniformOfFintype G).map (fun k => (m', k * m')));
    joint (m, c) / (∑' x : G, joint (x, c)) = messageDist m := by
  intro joint
  -- The joint probability of `(x, c)` is `P(M = x) / |G|`, because given `M = x` the key is
  -- uniquely determined as `c * x⁻¹` and the key is uniform.
  have hpt : ∀ x : G, joint (x, c) = messageDist x * (Fintype.card G : ℝ≥0∞)⁻¹ := by
    intro x
    have hmap : ∀ a : G,
        (PMF.map (fun k => (a, k * a)) (PMF.uniformOfFintype G)) (x, c)
          = if x = a then (Fintype.card G : ℝ≥0∞)⁻¹ else 0 := by
      intro a
      rw [PMF.map_apply]
      by_cases hxa : x = a
      · subst hxa
        rw [if_pos rfl, tsum_eq_single (c * x⁻¹)]
        · rw [PMF.uniformOfFintype_apply]
          have hxc : (x, c) = (x, (c * x⁻¹) * x) := by
            congr 1; rw [mul_assoc, inv_mul_cancel, mul_one]
          rw [if_pos hxc]
        · intro k hk
          rw [if_neg]
          intro h
          apply hk
          have h2 : c = k * x := (Prod.ext_iff.mp h).2
          rw [h2, mul_assoc, mul_inv_cancel, mul_one]
      · rw [if_neg hxa]
        refine (tsum_congr (fun k => ?_)).trans tsum_zero
        exact if_neg (fun h => hxa (Prod.ext_iff.mp h).1)
    show (messageDist.bind (fun m' => (PMF.uniformOfFintype G).map (fun k => (m', k * m')))) (x, c)
        = messageDist x * (Fintype.card G : ℝ≥0∞)⁻¹
    rw [PMF.bind_apply, tsum_eq_single x]
    · rw [hmap x, if_pos rfl]
    · intro a ha
      rw [hmap a, if_neg (fun h => ha h.symm), mul_zero]
  -- Hence the ciphertext marginal is `1/|G|`: the ciphertext is uniform.
  have hden : (∑' x : G, joint (x, c)) = (Fintype.card G : ℝ≥0∞)⁻¹ := by
    simp_rw [hpt]
    rw [ENNReal.tsum_mul_right, PMF.tsum_coe, one_mul]
  rw [hpt m, hden]
  have h0 : (Fintype.card G : ℝ≥0∞)⁻¹ ≠ 0 :=
    ENNReal.inv_ne_zero.mpr (ENNReal.natCast_ne_top _)
  have htop : (Fintype.card G : ℝ≥0∞)⁻¹ ≠ ⊤ :=
    ENNReal.inv_ne_top.mpr (by exact_mod_cast Fintype.card_ne_zero)
  rw [mul_div_assoc, ENNReal.div_self h0 htop, mul_one]

is that in a four-term exact sequence A → B → C → D where A and D vanish,
the middle map B → C is bijective (and hence an isomorphism).

This is the algebraic engine behind the computation π₃(S²) ≅ ℤ via the Hopf
fibration: the vanishing of π₃(S¹), π₂(S¹), and π₂(S³) forces the map
π₃(S³) → π₃(S²) to be an isomorphism.
